// Converte o conteúdo lido de um código em um EAN-13 válido (13 dígitos).
// Retorna null se a leitura não for um EAN-13 confiável (ex.: leitura incompleta).
//
// Caixas de remédio no Brasil costumam ter EAN-13 (começa com 789 ou 790).
// Algumas também trazem um código 2D (DataMatrix GS1) que começa com "01"
// seguido do GTIN de 14 dígitos; tratamos esse caso também.
 
function hasValidCheckDigit(ean13: string): boolean {
  let sum = 0;
  for (let i = 0; i < 12; i++) {
    const digit = Number(ean13[i]);
    sum += i % 2 === 0 ? digit : digit * 3;
  }
  const check = (10 - (sum % 10)) % 10;
  return check === Number(ean13[12]);
}
 
export function normalizeBarcode(raw: string): string | null {
  const text = raw.trim();
 
  // GS1: identificador "01" + GTIN-14 (pode vir com parênteses ou espaços)
  const gs1 = text.replace(/[()\s]/g, '').match(/^01(\d{14})/);
  let digits = gs1 ? gs1[1] : text.replace(/\D/g, '');
 
  // GTIN-14 que é um EAN-13 com um zero na frente
  if (digits.length === 14 && digits.startsWith('0')) {
    digits = digits.slice(1);
  }
 
  if (digits.length !== 13) return null;
  return hasValidCheckDigit(digits) ? digits : null;
}
