// Controle de quantidade de comprimidos (lógica pura, sem depender do celular).
import type { AlarmConfig } from '@/utils/schedule';
 
// Avisa para comprar mais quando os comprimidos restantes NÃO dão para este número de dias.
export const LOW_STOCK_DAYS = 7;
 
// Quantos comprimidos são necessários para durar LOW_STOCK_DAYS dias: abaixo disso, está acabando.
export function lowThreshold(intervalHours: number, pillsPerDose: number): number {
  const perDay = (24 / intervalHours) * pillsPerDose;
  return Math.max(pillsPerDose, Math.ceil(perDay * LOW_STOCK_DAYS));
}
 
function endOfLocalDay(iso: string): number {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d, 23, 59, 59, 999).getTime();
}
 
// Quantas doses estavam marcadas no intervalo (from, to], respeitando a data final.
// O app considera que cada alarme foi atendido (ainda não existe o botão "Tomei").
export function dosesBetween(cfg: AlarmConfig, from: Date, to: Date): number {
  const anchor = new Date(cfg.anchor).getTime();
  const step = cfg.intervalHours * 3600000;
  const end = cfg.endDate ? endOfLocalDay(cfg.endDate) : Infinity;
 
  const last = Math.min(to.getTime(), end);
  const firstIndex = Math.max(0, Math.floor((from.getTime() - anchor) / step) + 1);
  if (last < anchor + firstIndex * step) return 0;
  return Math.floor((last - anchor) / step) - firstIndex + 1;
}
 
export type StockPlan =
  | { kind: 'none' }
  | { kind: 'now'; remaining: number }
  | { kind: 'later'; at: Date; remaining: number };
 
// Decide quando avisar que o remédio está acabando:
// - 'now': já não dá para o prazo agora;
// - 'later': vai passar do limite na dose de `at` (o aviso é agendado para esse horário);
// - 'none': não vai acabar enquanto durar o alarme.
export function planStockAlert(
  cfg: AlarmConfig,
  stock: number,
  pillsPerDose: number,
  now: Date,
): StockPlan {
  const threshold = lowThreshold(cfg.intervalHours, pillsPerDose);
  if (stock < threshold) return { kind: 'now', remaining: stock };
 
  const anchor = new Date(cfg.anchor).getTime();
  const step = cfg.intervalHours * 3600000;
  const end = cfg.endDate ? endOfLocalDay(cfg.endDate) : Infinity;
 
  let k = Math.max(0, Math.ceil((now.getTime() + 5000 - anchor) / step));
  let remaining = stock;
  for (let i = 0; i < 20000; i++, k++) {
    const t = anchor + k * step;
    if (t > end) return { kind: 'none' };
    remaining -= pillsPerDose;
    if (remaining < threshold) {
      // 1 minuto depois da dose, para o aviso não competir com o alarme.
      return { kind: 'later', at: new Date(t + 60000), remaining: Math.max(0, remaining) };
    }
  }
  return { kind: 'none' };
}
